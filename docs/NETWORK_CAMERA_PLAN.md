## 16. 附录 D: Sprint 0 一键验收脚本 `tools/sprint0_smoke.sh`

> Sprint 0 完成的唯一硬指标: 跑完这个脚本全部 PASS. 失败则 Sprint 0 未完, 不进入 Sprint 1.

```bash
#!/usr/bin/env bash
#
# tools/sprint0_smoke.sh
# Sprint 0 验收: 6 路 IP cam RTSP → 6 路 DMA-BUF → 6 路 2×3 拼接出图
#
# 用法: ssh 板子, bash tools/sprint0_smoke.sh
#

set -uo pipefail

PROJECT_ROOT="${PROJECT_ROOT:-/home/rocktech/Projects/new-4k-stitch}"
cd "${PROJECT_ROOT}"

PASS=0; FAIL=0; WARN=0
ok()    { echo "[PASS] $*"; PASS=$((PASS+1)); }
warn()  { echo "[WARN] $*"; WARN=$((WARN+1)); }
fail()  { echo "[FAIL] $*"; FAIL=$((FAIL+1)); }

bar() { echo ""; echo "===== $* ====="; }

# ---------- 1. yaml + INPUT_SOURCE_MODE ----------
bar "1. camera_sources.yaml + INPUT_SOURCE_MODE"
if [ ! -f params/camera_sources.yaml ]; then fail "yaml not found"; else ok "yaml exists"; fi
echo "  INPUT_SOURCE_MODE=${INPUT_SOURCE_MODE:-<unset>}"
N=$(grep -c 'type: rtsp' params/camera_sources.yaml || true)
if [ "${N}" = "6" ]; then ok "6 type=rtsp entries"; else fail "expected 6 rtsp entries, got ${N}"; fi

# ---------- 2. 6 路 TCP port 可达 ----------
bar "2. 6 路 IP cam TCP/554 可达性"
reachable=0
for ip in $(grep -E 'uri:' params/camera_sources.yaml | grep -oE '192.168.10.[0-9]+' | sort -u); do
    if timeout 3 bash -c "</dev/tcp/${ip}/554" 2>/dev/null; then
        ok "TCP/554 reachable on ${ip}"
        reachable=$((reachable+1))
    else
        warn "TCP/554 unreachable on ${ip}"
    fi
done
if [ "${reachable}" -ge 1 ]; then ok "${reachable} cams reachable"; else fail "no cam reachable"; fi

# ---------- 3. gst-launch 单路烟测 (拉 5 帧即停, 不耗带宽) ----------
bar "3. 单路 gst-launch 烟测 (5 帧)"
first_uri=$(grep -E 'uri:' params/camera_sources.yaml | head -1 | awk '{print $2}')
smoke_out=$(timeout 12 gst-launch-1.0 -v \
    rtspsrc location="${first_uri}" latency=120 protocols=4 \
    ! rtph264depay ! h264parse \
    ! mppvideodec dma-feature=true format=NV12 \
    ! "video/x-raw(memory:DMABuf),format=NV12" \
    ! fakesink num-buffers=5 2>&1 || true)
if echo "${smoke_out}" | grep -q "ERROR"; then
    fail "gst-launch ERROR:"
    echo "${smoke_out}" | grep -E "ERROR" | head -5 | sed 's/^/    /'
else
    ok "gst-launch 5 帧无 ERROR"
fi
if echo "${smoke_out}" | grep -qE "video/x-raw.*memory:DMABuf"; then
    ok "mppvideodec 输出 DMA-BUF (zero-copy 路径生效)"
else
    fail "mppvideodec 输出非 DMA-BUF (违反零拷贝约束)"
fi

# ---------- 4. image-stitching 编译 ----------
bar "4. image-stitching 编译"
mkdir -p build && cd build
if cmake -DCMAKE_BUILD_TYPE=Release .. 2>&1 | grep -qE "Error|error"; then
    fail "cmake 配置失败"; cd ..
else
    if make -j$(nproc) 2>&1 | grep -qE "Error|error:"; then
        fail "make 编译失败"; cd ..
    else
        ok "cmake + make 通过"; cd ..
    fi
fi

# ---------- 5. 跑 image-stitching 30 秒, 检查日志 ----------
bar "5. image-stitching 运行 30 秒, 解码 & stitch FPS"
rm -f /tmp/stitch_status.json
( ./build/image-stitching > logs/sprint0.log 2>&1 ) &
PID=$!
sleep 30
if ! kill -0 ${PID} 2>/dev/null; then
    fail "image-stitching 在 30s 内退出 (PID=${PID} 已死)"
    tail -30 logs/sprint0.log | sed 's/^/    /'
else
    ok "image-stitching 在 30s 后仍存活 (PID=${PID})"
fi

# 检查 decoder_perf
grep "decoder_perf" logs/sprint0.log | tail -12 | sed 's/^/    /'
all_decode_ok=true
for i in 0 1 2 3 4 5; do
    line=$(grep "decoder_perf ${i}" logs/sprint0.log | tail -1 || true)
    fps=$(echo "${line}" | grep -oE "fps=[0-9.]+" | head -1 | sed 's/fps=//')
    if [ -z "${fps}" ] || [ "$(echo "${fps} | awk '{print ($1 < 20)}')" = "1" ]; then
        warn "cam${i} fps=${fps:-?} (< 20)"
        all_decode_ok=false
    else
        ok "cam${i} fps=${fps}"
    fi
done

# 检查 DIAG 段 6 路都有
diag_count=$(grep -c "gst_mpp_decoder.*DIAG.*layout=1" logs/sprint0.log || true)
if [ "${diag_count}" -ge 6 ]; then
    ok "${diag_count} DIAG 段 layout=1 (单 fd, NV12 零拷贝)"
else
    warn "只有 ${diag_count} DIAG 段 layout=1 (期望 ≥ 6)"
fi

kill ${PID} 2>/dev/null || true; wait ${PID} 2>/dev/null || true

# ---------- 6. CameraPage JSON ----------
bar "6. CameraPage /tmp/stitch_status.json"
if [ -f /tmp/stitch_status.json ]; then
    if grep -q "rtsp://" /tmp/stitch_status.json; then
        ok "status.json 含 rtsp:// URL"
    else
        fail "status.json 不含 rtsp:// URL (status_writer 没接 yaml)"
        cat /tmp/stitch_status.json | head -8 | sed 's/^/    /'
    fi
else
    fail "status.json 没生成 (status_writer 线程没起来)"
fi

# ---------- 总结 ----------
bar "SUMMARY"
echo "  PASS=${PASS}  WARN=${WARN}  FAIL=${FAIL}"
if [ "${FAIL}" -eq 0 ]; then
    echo "  ✅ Sprint 0 PASS — 进入 Sprint 1"
    exit 0
else
    echo "  ❌ Sprint 0 FAIL — 修 FAIL 项再跑"
    exit 1
fi
```

### 16.2 Sprint 0 失败常见模式 + 排查

| 症状 | 可能根因 | 排查 |
|---|---|---|
| TCP/554 unreachable | 摄像机没通电 / 网线松 / PoE 故障 | `ping 192.168.10.21`, 摄像机 web 后台 |
| gst-launch ERROR: "Could not open resource" | RTSP URL 路径错 (vendor 私有路径) | 摄像机 web 后台确认 URL, 重写 yaml |
| gst-launch 5 帧无 ERROR, image-stitching 0 帧 | yaml 解析崩, fallback 到 dataset | `grep "yaml" logs/sprint0.log` |
| decoder fps < 10 | 网卡驱动不行 / 摄像机帧率低于 30 | `ethtool eth0`, `gst-discoverer "${uri}"` |

| image-stitching 退出 | CMake 链接错误 / dpkg 缺失 `gstreamer1.0-plugins-good` | `dpkg -l | grep gstreamer1.0-plugins-good` |
